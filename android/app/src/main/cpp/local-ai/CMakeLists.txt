cmake_minimum_required(VERSION 3.22)
project(atlas_local_ai LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# APK debug signing must not imply an unoptimized inference engine.
if(ANDROID)
 add_compile_options(-O3)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
if(DEFINED ATLAS_LLAMA_SOURCE_DIR)
 set(llama_SOURCE_DIR "${ATLAS_LLAMA_SOURCE_DIR}")
else()
 include(FetchContent)
 FetchContent_Declare(llama URL "https://codeload.github.com/ggml-org/llama.cpp/tar.gz/a6ac41c757c066941766164439f6876f2deaab04" URL_HASH "SHA256=2696af3883b799a31102b1ce2c107212286d0e5219143dbfb8edc70cd52f17a2")
 FetchContent_GetProperties(llama)
 if(NOT llama_POPULATED)
  FetchContent_Populate(llama)
 endif()
endif()
add_subdirectory("${llama_SOURCE_DIR}" llama-build EXCLUDE_FROM_ALL)
add_library(atlas_core STATIC atlas_engine.cpp atlas_template.cpp
 "${llama_SOURCE_DIR}/common/jinja/lexer.cpp" "${llama_SOURCE_DIR}/common/jinja/parser.cpp"
 "${llama_SOURCE_DIR}/common/jinja/runtime.cpp" "${llama_SOURCE_DIR}/common/jinja/value.cpp"
 "${llama_SOURCE_DIR}/common/jinja/string.cpp" "${llama_SOURCE_DIR}/common/unicode.cpp" "${llama_SOURCE_DIR}/common/json.cpp")
target_include_directories(atlas_core PUBLIC . "${llama_SOURCE_DIR}/common" "${llama_SOURCE_DIR}/include" "${llama_SOURCE_DIR}/vendor/nlohmann" "${llama_SOURCE_DIR}/vendor")
target_link_libraries(atlas_core PUBLIC llama)
if(ANDROID)
 add_library(atlas_llama SHARED atlas_jni.cpp)
 target_link_libraries(atlas_llama PRIVATE atlas_core log android)
 target_link_options(atlas_llama PRIVATE "-Wl,-z,max-page-size=16384")
else()
 add_executable(atlas-cli atlas_cli.cpp)
 target_link_libraries(atlas-cli PRIVATE atlas_core)
 enable_testing()
 add_executable(atlas-core-test atlas_core_test.cpp)
 target_link_libraries(atlas-core-test PRIVATE atlas_core)
 target_compile_definitions(atlas-core-test PRIVATE ATLAS_TEMPLATE_DIR="${llama_SOURCE_DIR}/models/templates")
 add_test(NAME atlas-config-and-invalid-file COMMAND atlas-core-test)
endif()
